らくらくgetElementById(JavaScript用)
import pyperclip

output = ""

print('const {ID名} = document.getElementById("{ID名}");を出力します')
index = input("ID名を半角スペース区切りで入力:").split()

for loop in index:
    output += f'const {loop} = document.getElementById("{loop}");\n'

print(output)
pyperclip.copy(output)
print("クリップボードにコピーしました")
input("Enterで終了")
